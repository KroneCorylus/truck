cmake_minimum_required(VERSION 3.16)
project(truck_occt_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_path(OCCT_INCLUDE_DIR Standard_Version.hxx PATH_SUFFIXES opencascade REQUIRED)
add_executable(compare_occt main.cpp)
target_include_directories(compare_occt SYSTEM PRIVATE "${OCCT_INCLUDE_DIR}")
target_compile_options(compare_occt PRIVATE -Wall -Wextra -Wpedantic)
foreach(lib TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKFillet TKMesh TKXSBase TKDESTEP)
  find_library(OCCT_${lib} NAMES ${lib} HINTS "${OCCT_LIBRARY_DIR}" REQUIRED)
  target_link_libraries(compare_occt PRIVATE "${OCCT_${lib}}")
endforeach()
